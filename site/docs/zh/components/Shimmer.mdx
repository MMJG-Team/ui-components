# Shimmer 流光效果

Shimmer 是一个流光效果组件

## 代码演示

### 基本用法

```tsx preview
import React from "react";
import { Shimmer } from "@mmjg/ui-components";

const BasicDemo: React.FC = () => {
    return (
        <div
            style={{
                display: "flex",
                justifyContent: "center",
            }}
        >
            <Shimmer
                style={{
                    display: "flex",
                    justifyContent: "center",
                    width: "fit-content",
                    border: "1px dashed var(--rp-c-brand)",
                    borderRadius: "4px",
                    padding: "0px 20px",
                }}
            >
                Text Effect
            </Shimmer>
        </div>
    );
};

export default BasicDemo;
```
